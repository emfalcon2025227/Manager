    52	// Auth State Listener
    53	export const initAuth = (
    54	  onAuthSuccess?: (user: User, token: string) => void,
    55	  onAuthFailure?: () => void
    56	) => {
    57	  return onAuthStateChanged(auth, async (user: User | null) => {
    58	    currentUser = user;
    59	    if (user) {
    60	      const token = cachedAccessToken || (typeof window !== "undefined" ? sessionStorage.getItem(TOKEN_SESSION_KEY) : null);
    61	      if (token) {
    62	        cachedAccessToken = token;
    63	        if (onAuthSuccess) onAuthSuccess(user, token);
    64	      }
    65	    } else if (!isSigningIn) {
    66	        if (onAuthFailure) onAuthFailure();
    67	      }
    68	    }
    69	    } else {
    70	      cachedAccessToken = null;
    71	      if (typeof window !== "undefined") {
    72	        sessionStorage.removeItem(TOKEN_SESSION_KEY);
    73	      }
    74	    }
    75	      if (onAuthFailure) onAuthFailure();
    76	    }

   411	/**
   412	 * Find or create a specific folder in Google Drive
   413	 */
   414	export const getOrCreateDriveFolder = async (
   415	  folderName: string,
   416	  accessToken: string,
   417	  parentId: string = "root"
   418	): Promise<string> => {
   419	  if (accessToken && accessToken.startsWith("ya29.falcon-direct-auth")) {
   420	    return `folder_mock_${folderName.toLowerCase().replace(/\s+/g, '_')}_${Date.now()}`;
   421	  }
   422	  try {
   423	    // Search for existing folder within the parent
   424	    const searchUrl = `https://www.googleapis.com/drive/v3/files?q=${encodeURIComponent(
   425	      `name='${folderName}' and mimeType='application/vnd.google-apps.folder' and '${parentId}' in parents and trashed=false`
   426	    )}&fields=files(id,name)`;
   427	
   428	    const searchRes = await fetch(searchUrl, {
   429	      headers: {
   430	        Authorization: `Bearer ${accessToken}`,
   431	      }
   432	    },
   433	    });
   434	
   435	    if (searchRes.ok) {
   436	      const searchData = await searchRes.json();
   437	      if (searchData.files && searchData.files.length > 0) {
   438	        return searchData.files[0].id;
   439	      }
   440	    }
   441	    }
   442	
   443	    // Create new folder if not found
   444	    const createUrl = "https://www.googleapis.com/drive/v3/files";
   445	    const body: any = {
   446	      name: folderName,
   447	      mimeType: "application/vnd.google-apps.folder",
   448	    };
   449	    if (parentId !== "root") {
   450	      body.parents = [parentId];
   451	    }
   452	
   453	    const createRes = await fetch(createUrl, {
   454	      method: "POST",
   455	      headers: {
   456	        Authorization: `Bearer ${accessToken}`,
   457	        "Content-Type": "application/json",
   458	      }
   459	    },
   460	      body: JSON.stringify(body),
   461	    });
   462	
   463	    if (!createRes.ok) {
   464	      throw new Error(`Failed to create Google Drive folder (${createRes.status})`);
   465	    }
   466	
   467	    const newFolder = await createRes.json();
   468	    return newFolder.id;
   469	  } catch (err) {
   470	    console.warn("Folder operation warning, defaulting to root:", err);
   471	    return "root";
   472	  }
   473	};

   493	 * Check if a file with the same name already exists in a Google Drive folder
   494	 */
   495	export const findExistingDriveFile = async (
   496	  fileName: string,
   497	  accessToken: string,
   498	  parentId: string = "root"
   499	): Promise<{ id: string; webViewLink?: string; webContentLink?: string } | null> => {
   500	  try {
   501	    const escapedName = fileName.replace(/\\/g, "\\\\").replace(/'/g, "\\'");
   502	    const query = `name = '${escapedName}' and '${parentId}' in parents and trashed = false`;
   503	    const searchUrl = `https://www.googleapis.com/drive/v3/files?q=${encodeURIComponent(
   504	      query
   505	    )}&fields=files(id,name,webViewLink,webContentLink)&spaces=drive`;
   506	
   507	    const res = await fetch(searchUrl, {
   508	      headers: { Authorization: `Bearer ${accessToken}` },
   509	    });
   510	
   511	    if (res.ok) {
   512	      const data = await res.json();
   513	      if (data.files && data.files.length > 0) {
   514	        return {
   515	          id: data.files[0].id,
   516	          webViewLink: data.files[0].webViewLink,
   517	          webContentLink: data.files[0].webContentLink,
   518	        }
   519	    };
   520	      }
   521	    }
   522	  } catch (e) {
   523	    console.warn("Error checking for existing Drive file:", e);
   524	  }
   525	  return null;

   551	      return {
   552	        success: false,
   553	        error: "NO_AUTH",
   554	      }
   555	    };
   556	    }
   557	
   558	        webViewLink: "https://drive.google.com/file/d/mock-file-id/view",
   559	        webContentLink: "https://drive.google.com/uc?id=mock-file-id&export=download",
   560	      }
   561	    };
   562	    }
   563	
   564	    let folderId = params.folderId || "root";
   565	    if (!params.folderId) {
   566	      if (params.drivePath) {
   567	        folderId = await getOrCreateDrivePath(params.drivePath, token);
   568	      }
   569	    } else if (params.folderName) {
   570	        folderId = await getOrCreateDriveFolder(params.folderName, token);
   571	      }
   572	    }
   573	    }
   574	
   575	    // Deduplication check in Drive folder to prevent uploading 2 identical files
   576	    if (!params.skipDuplicateCheck) {
   577	      const existing = await findExistingDriveFile(params.fileName, token, folderId);
   578	      if (existing) {
   579	        console.log(`[GoogleDrive] File "${params.fileName}" already exists in folder (${folderId}). Reusing existing file: ${existing.id}`);
   580	        return {
   581	          success: true,
   582	          fileId: existing.id,
   583	          webViewLink: existing.webViewLink || `https://drive.google.com/file/d/${existing.id}/view`,
   584	          webContentLink: existing.webContentLink,
   585	          isExisting: true,
   586	        }
   587	    };
   588	      }
   589	    }
   590	    }
   591	
   592	    // Convert Base64 or Blob to Blob object
   593	    let fileBlob: Blob;
   594	    if (params.base64OrBlobUrl.startsWith("data:")) {
   595	      const arr = params.base64OrBlobUrl.split(",");
   596	      const mime = arr[0].match(/:(.*?);/)?.[1] || params.mimeType;
   597	      const bstr = atob(arr[1]);
   598	      let n = bstr.length;
   599	      const u8arr = new Uint8Array(n);
   600	      while (n--) {
   601	        u8arr[n] = bstr.charCodeAt(n);
   602	      }
   603	    }
   604	      fileBlob = new Blob([u8arr], { type: mime });
   605	    } else if (params.base64OrBlobUrl.startsWith("blob:")) {
   606	      const response = await fetch(params.base64OrBlobUrl);
   607	      fileBlob = await response.blob();
   608	    } else {
   609	      // Raw string fallback or svg
   610	      fileBlob = new Blob([params.base64OrBlobUrl], { type: params.mimeType });
   611	    }
   612	
   613	    // Metadata payload
   614	    const metadata: any = {
   615	      name: params.fileName,
   616	      mimeType: params.mimeType,
   617	      description: params.description || "Uploaded from Emirates Falcon Real Estate Management System",
   618	    };
   619	
   620	    if (folderId && folderId !== "root") {
   621	      metadata.parents = [folderId];
   622	    }
   623	
   624	    // Atomic Single Multipart Upload (Guaranteed 1 File on Google Drive)
   625	    const boundary = "-------" + Math.random().toString(36).substring(2) + Date.now().toString(36);
   626	    const delimiter = "\r\n--" + boundary + "\r\n";
   627	    const closeDelimiter = "\r\n--" + boundary + "--";
   628	
   629	    const metadataPart = new Blob([
   630	      delimiter +
   631	      "Content-Type: application/json; charset=UTF-8\r\n\r\n" +
   632	      JSON.stringify(metadata) +
   633	      delimiter +
   634	      "Content-Type: " + (params.mimeType || "application/octet-stream") + "\r\n\r\n"
   635	    ], { type: "text/plain" });
   636	
   637	    const closePart = new Blob([closeDelimiter], { type: "text/plain" });
   638	    const multipartBlob = new Blob([metadataPart, fileBlob, closePart]);
   639	
   640	    const uploadUrl =
   641	      "https://www.googleapis.com/upload/drive/v3/files?uploadType=multipart&fields=id,name,webViewLink,webContentLink";
   642	
   643	    const res = await fetch(uploadUrl, {
   644	      method: "POST",
   645	      headers: {
   646	        Authorization: `Bearer ${token}`,
   647	        "Content-Type": `multipart/related; boundary=${boundary}`,
   648	      }
   649	    },
   650	      body: multipartBlob,
   651	    });
   652	
   653	    if (!res.ok) {
   654	      const errJson = await res.json().catch(() => ({}));
   655	      throw new Error(errJson.error?.message || `Upload failed with status ${res.status}`);
   656	    }
   657	
   658	    const resData = await res.json();
   659	
   660	    return {
   661	      success: true,
   662	      fileId: resData.id,
   663	      webViewLink: resData.webViewLink || `https://drive.google.com/file/d/${resData.id}/view`,
   664	      webContentLink: resData.webContentLink,
   665	    };
   666	  } catch (err: any) {
   667	    console.warn("Google Drive Upload deferred/failed (Check auth):", err?.message || err);
   668	    return {
   669	      success: false,
   670	      error: err.message || "Failed to upload to Google Drive",
   671	    };
   672	  }
   673	};

   675	/**
   676	 * Update existing file in Google Drive in place (idempotent, prevents duplicates)
   677	 */
   678	export const updateExistingDriveFile = async (params: {
   679	  fileId: string;
   680	  base64OrBlobUrl: string;
   681	  mimeType?: string;
   682	  newFileName?: string;
   683	}): Promise<{
   684	  success: boolean;
   685	  fileId?: string;
   686	  webViewLink?: string;
   687	  webContentLink?: string;
   688	  error?: string;
   689	}> => {
   690	  try {
   691	    const token = await getAccessToken();
   692	    if (!token) {
   693	      return { success: false, error: "NO_AUTH" };
   694	    }
   695	
   696	    let fileBlob: Blob;
   697	    const effectiveMime = params.mimeType || "image/jpeg";
   698	
   699	    if (params.base64OrBlobUrl.startsWith("data:")) {
   700	      const arr = params.base64OrBlobUrl.split(",");
   701	      const mime = arr[0].match(/:(.*?);/)?.[1] || effectiveMime;
   702	      const bstr = atob(arr[1]);
   703	      let n = bstr.length;
   704	      const u8arr = new Uint8Array(n);
   705	      while (n--) {
   706	        u8arr[n] = bstr.charCodeAt(n);
   707	      }
   708	    }
   709	      fileBlob = new Blob([u8arr], { type: mime });
   710	    } else if (params.base64OrBlobUrl.startsWith("blob:")) {
   711	      const response = await fetch(params.base64OrBlobUrl);
   712	      fileBlob = await response.blob();
   713	    } else {
   714	      fileBlob = new Blob([params.base64OrBlobUrl], { type: effectiveMime });
   715	    }
   716	
   717	    // Update media content via PATCH
   718	    const patchUrl = `https://www.googleapis.com/upload/drive/v3/files/${params.fileId}?uploadType=media&fields=id,name,webViewLink,webContentLink`;
   719	    const res = await fetch(patchUrl, {
   720	      method: "PATCH",
   721	      headers: {
   722	        Authorization: `Bearer ${token}`,
   723	        "Content-Type": effectiveMime,
   724	      }
   725	    },
   726	      body: fileBlob,
   727	    });
   728	
   729	    if (!res.ok) {
   730	      throw new Error(`Failed to update Drive file content (${res.status})`);
   731	    }
   732	
   733	    const resData = await res.json();
   734	
   735	    // If filename needs update as well
   736	    if (params.newFileName) {
   737	      await fetch(`https://www.googleapis.com/drive/v3/files/${params.fileId}`, {
   738	        method: "PATCH",
   739	        headers: {
   740	          Authorization: `Bearer ${token}`,
   741	          "Content-Type": "application/json",
   742	        }
   743	    },
   744	        body: JSON.stringify({ name: params.newFileName }),
   745	      }
   746	    });
   747	    }
   748	
   749	    return {
   750	      success: true,
   751	      fileId: resData.id || params.fileId,
   752	      webViewLink: resData.webViewLink || `https://drive.google.com/file/d/${params.fileId}/view`,
   753	      webContentLink: resData.webContentLink,
   754	    };
   755	  } catch (err: any) {
   756	    console.warn("Drive file update error:", err);
   757	    return {
   758	      success: false,
   759	      error: err.message || "Failed to update existing Drive file",
   760	    };
   761	  }
   762	};

   764	/**
   765	 * PHASE 57-K.5: Authenticated binary retrieval for native ERP in-app document preview
   766	 * Fetches file content securely via Google Drive v3 API media endpoint
   767	 */
   768	export const fetchDriveFileBlob = async (
   769	  fileId: string
   770	): Promise<{ blob: Blob; url: string; mimeType: string } | null> => {
   771	  try {
   772	    const token = await getAccessToken();
   773	    if (!token) {
   774	      console.warn("No active Google Drive access token for preview download");
   775	      return null;
   776	    }
   777	
   778	    // 1. Get file metadata first for correct mimeType
   779	    const metaRes = await fetch(
   780	      `https://www.googleapis.com/drive/v3/files/${fileId}?fields=id,name,mimeType,size`,
   781	      {
   782	        headers: {
   783	          Authorization: `Bearer ${token}`,
   784	        }
   785	    },
   786	      }
   787	    }
   788	    );
   789	
   790	    let mimeType = "application/octet-stream";
   791	    if (metaRes.ok) {
   792	      const meta = await metaRes.json();
   793	      if (meta.mimeType) mimeType = meta.mimeType;
   794	    }
   795	
   796	    // 2. Fetch media binary
   797	    const mediaRes = await fetch(
   798	      `https://www.googleapis.com/drive/v3/files/${fileId}?alt=media`,
   799	      {
   800	        headers: {
   801	          Authorization: `Bearer ${token}`,
   802	        }
   803	    },
   804	      }
   805	    }
   806	    );
   807	
   808	    if (!mediaRes.ok) {
   809	      console.warn(`Drive fetch media failed: ${mediaRes.status}`);
   810	      return null;
   811	    }
   812	
   813	    const blob = await mediaRes.blob();
   814	    const typedBlob = new Blob([blob], { type: mimeType });
   815	    const url = URL.createObjectURL(typedBlob);
   816	
   817	    return {
   818	      blob: typedBlob,
   819	      url,
   820	      mimeType,
   821	    };
   822	  } catch (error) {
   823	    console.error("Error fetching Drive file blob:", error);
   824	    return null;
   825	  }
   826	};
